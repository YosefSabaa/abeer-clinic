'use client'
import { useState, useEffect, useMemo } from 'react'
import Header from '@/components/Header'
import BottomNav from '@/components/BottomNav'
import Footer from '@/components/Footer'
import {
  getAdminDashboardData,
  upsertPatientAdmin,
  updateAppointmentStatus,
  addMedicalRecord,
  deleteMedicalRecord,
} from '@/lib/booking'

export default function AdminDashboardPage() {
  const [locale, setLocale] = useState('ar')
  const [mounted, setMounted] = useState(false)
  const [isAuthenticated, setIsAuthenticated] = useState(false)
  const [pinInput, setPinInput] = useState('')
  const [pinError, setPinError] = useState('')

  const [data, setData] = useState({ patients: [], appointments: [], medicalRecords: [] })
  const [loading, setLoading] = useState(false)
  const [activeTab, setActiveTab] = useState('patients')
  const [searchQuery, setSearchQuery] = useState('')
  const [statusFilter, setStatusFilter] = useState('all')
  const [toast, setToast] = useState(null)

  const [showPatientForm, setShowPatientForm] = useState(false)
  const [patientForm, setPatientForm] = useState({
    name: '',
    phone: '',
    age: '',
    age_unit: 'years',
    marital_status: '',
    chronic_notes: '',
  })

  const [selectedPatient, setSelectedPatient] = useState(null)
  const [recordForm, setRecordForm] = useState({
    patient_phone: '',
    patient_name: '',
    visit_date: new Date().toISOString().split('T')[0],
    symptoms: '',
    diagnosis: '',
    prescription: '',
    notes: '',
    next_visit: '',
  })

  const showNotification = (text, type = 'success') => {
    setToast({ text, type })
    setTimeout(() => setToast(null), 3500)
  }

  const loadDashboard = async () => {
    setLoading(true)
    try {
      const res = await getAdminDashboardData()
      setData(res)
    } catch (err) {
      showNotification(err.message || 'فشل تحميل البيانات', 'error')
    }
    setLoading(false)
  }

  useEffect(() => {
    const initialize = async () => {
      const savedLocale = localStorage.getItem('locale')

      if (savedLocale === 'ar' || savedLocale === 'en') {
        setLocale(savedLocale)
      }

      try {
        const response = await fetch('/api/admin/session', {
          cache: 'no-store',
        })

        const result = await response.json()

        if (result.authenticated) {
          setIsAuthenticated(true)
          await loadDashboard()
        }
      } catch {
        setIsAuthenticated(false)
      }

      setMounted(true)
    }

    initialize()
  }, [])

  useEffect(() => {
    if (!mounted) return
    localStorage.setItem('locale', locale)
    document.documentElement.lang = locale
    document.documentElement.dir = locale === 'ar' ? 'rtl' : 'ltr'
  }, [locale, mounted])

  const handlePinSubmit = async (e) => {
    e.preventDefault()
    setPinError('')

    try {
      const response = await fetch('/api/admin/login', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          pin: pinInput,
        }),
      })

      const result = await response.json()

      if (!response.ok) {
        throw new Error(result.error || 'رمز الدخول غير صحيح')
      }

      setIsAuthenticated(true)
      setPinInput('')
      await loadDashboard()
    } catch (error) {
      setPinError(error.message || 'تعذر تسجيل الدخول')
    }
  }

  const handleSavePatient = async (e) => {
    e.preventDefault()
    try {
      await upsertPatientAdmin(patientForm)
      showNotification('تم حفظ بيانات المريض بنجاح')
      setPatientForm({ name: '', phone: '', age: '', age_unit: 'years', marital_status: '', chronic_notes: '' })
      setShowPatientForm(false)
      await loadDashboard()
    } catch (err) {
      showNotification(err.message, 'error')
    }
  }

  const handleEditPatientClick = (patient) => {
    setPatientForm({
      name: patient.name || '',
      phone: patient.phone || '',
      age: patient.age || '',
      age_unit: patient.age_unit || 'years',
      marital_status: patient.marital_status || '',
      chronic_notes: patient.chronic_notes || '',
    })
    setShowPatientForm(true)
    window.scrollTo({ top: 0, behavior: 'smooth' })
  }

  const openPatientMedicalFile = (patient) => {
    setSelectedPatient(patient)
    setRecordForm({
      patient_phone: patient.phone,
      patient_name: patient.name,
      visit_date: new Date().toISOString().split('T')[0],
      symptoms: '',
      diagnosis: '',
      prescription: '',
      notes: '',
      next_visit: '',
    })
  }

  const handleAddRecord = async (e) => {
    e.preventDefault()
    try {
      await addMedicalRecord(recordForm)
      showNotification('تم إضافة الكشف والروشتة إلى التاريخ المرضي')
      setRecordForm((prev) => ({
        ...prev,
        symptoms: '',
        diagnosis: '',
        prescription: '',
        notes: '',
        next_visit: '',
      }))
      await loadDashboard()
    } catch (err) {
      showNotification(err.message, 'error')
    }
  }

  const handleDeleteRecord = async (id) => {
    if (!confirm('هل أنت متأكد من حذف هذا السجل المرضي؟')) return
    try {
      await deleteMedicalRecord(id)
      showNotification('تم حذف السجل')
      await loadDashboard()
    } catch (err) {
      showNotification(err.message, 'error')
    }
  }

  const handleStatusChange = async (id, newStatus) => {
    try {
      await updateAppointmentStatus(id, newStatus)
      showNotification('تم تحديث حالة الحجز')
      await loadDashboard()
    } catch (err) {
      showNotification(err.message, 'error')
    }
  }

  const filteredPatients = useMemo(() => {
    const q = searchQuery.trim().toLowerCase()
    if (!q) return data.patients
    return data.patients.filter(
      (p) => p.name?.toLowerCase().includes(q) || p.phone?.includes(q)
    )
  }, [data.patients, searchQuery])

  const filteredAppointments = useMemo(() => {
    const q = searchQuery.trim().toLowerCase()
    return data.appointments.filter((a) => {
      const matchesStatus = statusFilter === 'all' || a.status === statusFilter
      const matchesQuery =
        !q || a.patient_name?.toLowerCase().includes(q) || a.patient_phone?.includes(q)
      return matchesStatus && matchesQuery
    })
  }, [data.appointments, searchQuery, statusFilter])

  const filteredRecords = useMemo(() => {
    const q = searchQuery.trim().toLowerCase()
    if (!q) return data.medicalRecords
    return data.medicalRecords.filter(
      (r) =>
        r.patient_name?.toLowerCase().includes(q) ||
        r.patient_phone?.includes(q) ||
        r.diagnosis?.toLowerCase().includes(q)
    )
  }, [data.medicalRecords, searchQuery])

  const stats = useMemo(() => {
    return {
      totalPatients: data.patients.length,
      confirmedAppointments: data.appointments.filter((a) => a.status === 'confirmed').length,
      completedAppointments: data.appointments.filter((a) => a.status === 'completed').length,
      totalRecords: data.medicalRecords.length,
    }
  }, [data])

  if (!mounted) {
    return (
      <div className="min-h-screen bg-[#1A050A] flex items-center justify-center">
        <div className="w-8 h-8 border-2 border-pink-500 border-t-transparent rounded-full animate-spin" />
      </div>
    )
  }

  return (
    <div className="min-h-screen bg-[#1A050A] text-gray-200 pb-24 sm:pb-10">
      <Header locale={locale} setLocale={setLocale} />

      {toast && (
        <div
          className={`fixed top-16 left-1/2 -translate-x-1/2 z-50 px-4 py-2.5 rounded-xl text-xs font-bold shadow-lg border animate-slide-up ${
            toast.type === 'error'
              ? 'bg-red-950 text-red-300 border-red-700'
              : 'bg-green-950 text-green-300 border-green-700'
          }`}
        >
          {toast.text}
        </div>
      )}

      <main className="max-w-6xl mx-auto px-4 py-6 space-y-6">
        {!isAuthenticated ? (
          <div className="max-w-sm mx-auto bg-[#2A0A14] rounded-2xl p-6 border border-[#5A1D2E] shadow-card">
            <div className="w-12 h-12 rounded-2xl bg-purple-600/20 border border-purple-500/40 flex items-center justify-center text-purple-300 mx-auto mb-4">
              <svg className="w-6 h-6" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                <path strokeLinecap="round" strokeLinejoin="round" strokeWidth="2" d="M12 15v2m-6 4h12a2 2 0 002-2v-6a2 2 0 00-2-2H6a2 2 0 00-2 2v6a2 2 0 002 2zm10-10V7a4 4 0 00-8 0v4h8z" />
              </svg>
            </div>
            <h1 className="text-lg font-alexandria font-bold text-center text-white mb-1">
              لوحة تحكم العيادة
            </h1>
            <p className="text-xs text-gray-400 text-center mb-5">
              أدخل رمز الإدارة للمتابعة (الافتراضي: 1234)
            </p>
            <form onSubmit={handlePinSubmit} className="space-y-4">
              <input
                type="password"
                value={pinInput}
                onChange={(e) => setPinInput(e.target.value)}
                placeholder="••••"
                className="w-full text-center tracking-widest text-lg bg-[#3A0D1A] border border-[#5A1D2E] rounded-xl p-3 text-white focus:border-purple-500 outline-none"
              />
              {pinError && <p className="text-xs text-red-400 text-center">{pinError}</p>}
              <button
                type="submit"
                className="w-full bg-gradient-to-r from-pink-600 to-purple-600 text-white py-3 rounded-xl font-bold text-sm hover:opacity-90 transition"
              >
                دخول لوحة التحكم
              </button>
            </form>
          </div>
        ) : (
          <>
            <div className="flex flex-wrap items-center justify-between gap-3">
              <div>
                <h1 className="text-xl sm:text-2xl font-alexandria font-bold text-white">
                  إدارة المرضى والعيادة
                </h1>
                <p className="text-xs text-gray-400 mt-0.5">
                  دكتورة عبير الدالي — متابعة الحجوزات والسجلات الطبية
                </p>
              </div>
              <div className="flex items-center gap-2">
                <button
                  onClick={() => {
                    setPatientForm({ name: '', phone: '', age: '', age_unit: 'years', marital_status: '', chronic_notes: '' })
                    setShowPatientForm(!showPatientForm)
                  }}
                  className="bg-gradient-to-r from-pink-600 to-purple-600 text-white px-4 py-2.5 rounded-xl text-xs font-bold hover:opacity-90 transition shadow-glow"
                >
                  {showPatientForm ? 'إغلاق النموذج' : '+ إضافة / تعديل مريض'}
                </button>
                <button
                  onClick={loadDashboard}
                  disabled={loading}
                  className="bg-[#2A0A14] hover:bg-[#3A0D1A] border border-[#5A1D2E] text-gray-200 px-3.5 py-2.5 rounded-xl text-xs font-bold transition"
                >
                  {loading ? '...' : 'تحديث'}
                </button>
              </div>
            </div>

            {showPatientForm && (
              <form
                onSubmit={handleSavePatient}
                className="bg-[#2A0A14] rounded-2xl p-5 border border-pink-500/40 shadow-card space-y-4 animate-slide-up"
              >
                <h2 className="font-bold text-sm text-pink-300">بيانات ملف المريض</h2>
                <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
                  <div>
                    <label className="block text-[11px] text-gray-400 mb-1">اسم المريض *</label>
                    <input
                      type="text"
                      required
                      value={patientForm.name}
                      onChange={(e) => setPatientForm({ ...patientForm, name: e.target.value })}
                      className="w-full bg-[#3A0D1A] border border-[#5A1D2E] rounded-xl p-2.5 text-sm text-white outline-none focus:border-pink-500"
                      placeholder="الاسم الكامل"
                    />
                  </div>

                    <label className="block text-[11px] text-gray-400 mb-1">رقم الهاتف *</label>

import { supabaseAdmin } from '@/lib/supabase-admin'
import { sendAppointmentTemplate } from '@/lib/whatsapp/send-template'

function normalizePhone(raw) {
  let phone = String(raw || '').replace(/[\s\-()+]/g, '')

  if (phone.startsWith('0020')) {
    phone = '0' + phone.slice(4)
  } else if (phone.startsWith('+20')) {
    phone = '0' + phone.slice(3)
  } else if (/^1\d{9}$/.test(phone)) {
    phone = '0' + phone
  }

  return phone
}

export const runtime = 'nodejs'

export async function POST(request) {
  try {
    const body = await request.json()

    const name = String(body.name || '').trim()
    const phone = normalizePhone(body.phone)
    const date = String(body.date || '')
    const time = String(body.time || '')
    const maritalStatus = body.maritalStatus || null
    const age = body.age ? parseInt(body.age, 10) : null
    const ageUnit = body.ageUnit || 'years'
    const visitType = String(body.visitType || 'كشف')

    if (!name || !/^01\d{9}$/.test(phone) || !date || !time) {
      return Response.json(
        { success: false, message: 'يرجى مراجعة بيانات الحجز' },
        { status: 400 }
      )
    }

    const { data, error } = await supabaseAdmin.rpc(
      'book_appointment',
      {
        p_name: name,
        p_phone: phone,
        p_date: date,
        p_time: time,
        p_marital_status: maritalStatus,
        p_age: age,
        p_age_unit: ageUnit,
      }
    )

    if (error) {
      console.error('Booking RPC error:', error)

      return Response.json(
        { success: false, message: error.message },
        { status: 400 }
      )
    }

    let notificationSent = false

    if (data?.success) {
      try {
        await sendAppointmentTemplate({
          phone,
          name,
          date,
          time,
          visitType,
        })

        notificationSent = true
      } catch (notificationError) {
        console.error(
          'Appointment WhatsApp notification failed:',
          notificationError.message
        )
      }
    }

    return Response.json({
      ...data,
      whatsappNotification: notificationSent
        ? 'sent'
        : 'not_sent',
    })
  } catch (error) {
    console.error('Booking API error:', error)

    return Response.json(
      { success: false, message: 'حدث خطأ أثناء الحجز' },
      { status: 500 }
    )
  }
}

export const runtime = 'nodejs'

export async function GET(request) {
  if (!(await requireAdmin())) {
    return Response.json(
      { error: 'Unauthorized' },
      { status: 401 }
    )
  }

  const { searchParams } = new URL(request.url)
  const conversationId = searchParams.get('conversation_id')

  if (conversationId) {
    const { data, error } = await supabaseAdmin
      .from('whatsapp_messages')
      .select('*')
      .eq('conversation_id', conversationId)
      .order('created_at', { ascending: true })
      .limit(200)

    if (error) {
      console.error('Load messages error:', error)

      return Response.json(
        { error: error.message },
        { status: 500 }
      )
    }

    return Response.json({
      messages: data || [],
    })
  }

  const { data, error } = await supabaseAdmin
    .from('whatsapp_conversations')
    .select('*')
    .order('last_message_at', { ascending: false })
    .limit(100)

  if (error) {
    console.error('Load conversations error:', error)

    return Response.json(
      { error: error.message },
      { status: 500 }
    )
  }

  return Response.json({
    conversations: data || [],
  })
}

import { createHmac, timingSafeEqual } from 'node:crypto'
import { supabaseAdmin } from '@/lib/supabase-admin'

export const runtime = 'nodejs'

function safeEqual(a, b) {
  const x = Buffer.from(String(a || ''))
  const y = Buffer.from(String(b || ''))

  if (x.length !== y.length) return false

  return timingSafeEqual(x, y)
}

function getMessageContent(message) {
  switch (message.type) {
    case 'text':
      return message.text?.body || ''

    case 'image':
      return '[صورة]'

    case 'audio':
      return '[رسالة صوتية]'

    case 'video':
      return '[فيديو]'

    case 'document':
      return `[ملف] ${message.document?.filename || ''}`

    case 'location':
      return '[موقع جغرافي]'

    case 'sticker':
      return '[ملصق]'

    case 'interactive':
      return (
        message.interactive?.button_reply?.title ||
        message.interactive?.list_reply?.title ||
        '[رسالة تفاعلية]'
      )

    case 'button':
      return message.button?.text || '[زر]'

    default:
      return '[رسالة غير نصية]'
  }
}

export async function GET(request) {
  const url = new URL(request.url)

  const mode = url.searchParams.get('hub.mode')
  const token = url.searchParams.get('hub.verify_token')
  const challenge = url.searchParams.get('hub.challenge')

  const expected = process.env.WHATSAPP_VERIFY_TOKEN

  if (
    mode === 'subscribe' &&
    expected &&
    token &&
    challenge !== null &&
    safeEqual(token, expected)
  ) {
    return new Response(challenge, {
      status: 200,
      headers: { 'Content-Type': 'text/plain' },
    })
  }

  return new Response('Verification failed', { status: 403 })
}

export async function POST(request) {
  try {
    const appSecret = process.env.WHATSAPP_APP_SECRET

    if (!appSecret) {
      return Response.json(
        { error: 'Webhook is not configured' },
        { status: 500 }
      )
    }

    const rawBody = await request.text()
    const signature = request.headers.get('x-hub-signature-256')

    if (!signature?.startsWith('sha256=')) {
      return Response.json(
        { error: 'Invalid signature' },
        { status: 401 }
      )
    }

    const expectedSignature =
      'sha256=' +
      createHmac('sha256', appSecret)
        .update(rawBody)
        .digest('hex')

    if (!safeEqual(signature, expectedSignature)) {
      return Response.json(
        { error: 'Signature verification failed' },
        { status: 401 }
      )
    }

    const payload = JSON.parse(rawBody)

    for (const entry of payload.entry || []) {
      for (const change of entry.changes || []) {
        if (change.field !== 'messages') continue

        const value = change.value || {}

        // تحديث حالات الرسائل الصادرة
        for (const status of value.statuses || []) {
          const firstError = status.errors?.[0] || null

          const statusAt = status.timestamp
            ? new Date(Number(status.timestamp) * 1000).toISOString()
            : new Date().toISOString()

          const updateData = {
            status: status.status,
            status_at: statusAt,
            error_code: firstError?.code
              ? String(firstError.code)
              : null,
            error_title: firstError?.title || null,
            error_message: firstError?.message || null,
            error_details: firstError?.error_data?.details || null,
            raw_status: status,
          }

          const { error } = await supabaseAdmin
            .from('whatsapp_messages')
            .update(updateData)
            .eq('wa_message_id', status.id)

          if (error) {
            console.error('WhatsApp status update error:', {
              messageId: status.id,
              error,
            })
          }
        }

        // استقبال الرسائل الجديدة
        for (const message of value.messages || []) {
          const phone = message.from

          if (!phone) continue

          const content = getMessageContent(message)

          const contact = (value.contacts || []).find(
            (item) => item.wa_id === phone
          )

          const patientName = contact?.profile?.name || null
          const now = new Date().toISOString()

          const { data: conversation, error: conversationError } =
            await supabaseAdmin
              .from('whatsapp_conversations')
              .upsert(
                {
                  phone,
                  patient_name: patientName,
                  last_message: content,
                  last_message_at: now,
                  last_inbound_at: now,
                  updated_at: now,
                },
                { onConflict: 'phone' }
              )
              .select('id')
              .single()

          if (conversationError || !conversation) {
            console.error(
              'Conversation save error:',
              conversationError
            )
            continue
          }

          const { error: messageError } = await supabaseAdmin
            .from('whatsapp_messages')
            .insert({
              conversation_id: conversation.id,
              wa_message_id: message.id,
              direction: 'incoming',
              message_type: message.type || 'unknown',
              content,
              status: 'received',
              status_at: now,
            })

          // منع احتساب الرسالة مرتين عند إعادة إرسال Webhook
          if (messageError?.code === '23505') {
            continue
          }

          if (messageError) {
            console.error('Message save error:', messageError)
            continue
          }

          await supabaseAdmin.rpc(
            'increment_whatsapp_unread',
            { p_conversation_id: conversation.id }
          )
        }
      }
    }

    return Response.json({ received: true })
  } catch (error) {
    console.error('WhatsApp webhook error:', error)

    return Response.json(
      { error: 'Webhook processing failed' },
      { status: 500 }
    )
  }
}

export function normalizePhone(raw) {
  if (!raw) return ''
  let cleaned = String(raw).replace(/[\s\-()]/g, '').trim()
  if (cleaned.startsWith('+20')) cleaned = '0' + cleaned.slice(3)
  else if (cleaned.startsWith('0020')) cleaned = '0' + cleaned.slice(4)
  else if (/^1[0-9]{9}$/.test(cleaned)) cleaned = '0' + cleaned
  return cleaned
}

    .eq('phone', cleanPhone)

  // إنشاء ملف للمريض تلقائياً إذا دخل برقم جديد لأول مرة
  if (!patient) {

      .from('patients')
      .insert({ phone: cleanPhone, name: fallbackName })
      .select()
      .single()

    if (createError) throw new Error(createError.message)
    patient = created
  } else if (optionalName.trim() && patient.name.startsWith('مريض (')) {
    const { data: updated } = await supabase
      .from('patients')
      .update({ name: optionalName.trim() })
      .eq('id', patient.id)
      .select()
      .single()
    if (updated) patient = updated
  }

    .from('appointments')
    .select('*')
    .eq('patient_phone', cleanPhone)
    .order('appointment_date', { ascending: false })

    .from('medical_records')
    .select('*')
    .eq('patient_phone', cleanPhone)
    .order('visit_date', { ascending: false })
    .order('created_at', { ascending: false })

import { cookies } from 'next/headers'
import {
  createHmac,
  randomBytes,
  timingSafeEqual,
} from 'node:crypto'

const COOKIE_NAME = 'clinic_admin_session'
const SESSION_SECONDS = 60 * 60 * 8

function getSecret() {
  const secret = process.env.WHATSAPP_ADMIN_SESSION_SECRET

  if (!secret) {
    throw new Error('Missing WHATSAPP_ADMIN_SESSION_SECRET')
  }

  return secret
}

function sign(value) {
  return createHmac('sha256', getSecret())
    .update(value)
    .digest('hex')
}

function safeEqual(a, b) {
  const first = Buffer.from(String(a))
  const second = Buffer.from(String(b))

  if (first.length !== second.length) return false

  return timingSafeEqual(first, second)
}

export function verifyAdminPin(pin) {
  const configuredPin = process.env.ADMIN_PIN

  if (!configuredPin || !pin) return false

  return safeEqual(pin, configuredPin)
}

export async function createAdminSession() {
  const expiresAt = Date.now() + SESSION_SECONDS * 1000
  const nonce = randomBytes(24).toString('hex')

  const payload = `${expiresAt}.${nonce}`
  const signature = sign(payload)

  const cookieStore = await cookies()

  cookieStore.set(
    COOKIE_NAME,
    `${payload}.${signature}`,
    {
      httpOnly: true,
      secure: process.env.NODE_ENV === 'production',
      sameSite: 'strict',
      path: '/',
      maxAge: SESSION_SECONDS,
    }
  )
}

export async function isAdminAuthenticated() {
  const cookieStore = await cookies()
  const value = cookieStore.get(COOKIE_NAME)?.value

  if (!value) return false

  const parts = value.split('.')

  if (parts.length !== 3) return false

  const [expiresAt, nonce, signature] = parts
  const payload = `${expiresAt}.${nonce}`

  if (!/^\d+$/.test(expiresAt)) return false
  if (Number(expiresAt) < Date.now()) return false

  return safeEqual(signature, sign(payload))

function normalizeWhatsAppPhone(raw) {
  let phone = String(raw || '').replace(/\D/g, '')

  if (phone.startsWith('0020')) {
    phone = phone.slice(2)
  } else if (phone.startsWith('0')) {
    phone = '20' + phone.slice(1)
  } else if (phone.startsWith('1') && phone.length === 10) {
    phone = '20' + phone
  }

  if (!/^20\d{10}$/.test(phone)) {
    throw new Error('Invalid Egyptian WhatsApp phone number')
  }

  return phone
}

export async function sendAppointmentTemplate({
  phone,
  name,
  date,
  time,
  visitType,
}) {
  const token = process.env.WHATSAPP_ACCESS_TOKEN
  const phoneNumberId = process.env.WHATSAPP_PHONE_NUMBER_ID
  const version = process.env.WHATSAPP_API_VERSION

  if (!token || !phoneNumberId || !version) {
    throw new Error('WhatsApp API environment variables are missing')
  }

  const to = normalizeWhatsAppPhone(phone)

  const parameters = [
    name,
    DOCTOR.name,
    date,
    time,
    visitType || 'كشف',
    '360 جنيه',
    DOCTOR.address,
  ]

  const response = await fetch(
    `https://graph.facebook.com/${version}/${phoneNumberId}/messages`,
    {
      method: 'POST',
      headers: {
        Authorization: `Bearer ${token}`,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        messaging_product: 'whatsapp',
        recipient_type: 'individual',
        to,
        type: 'template',
        template: {
          name: 'appointment_confirmation_ar',
          language: {
            code: 'ar_EG',
          },
          components: [
            {
              type: 'body',
              parameters: parameters.map((value) => ({
                type: 'text',
                text: String(value ?? ''),
              })),
            },
          ],
        },
      }),
    }
  )

  const result = await response.json()

  if (!response.ok) {
    throw new Error(
      result.error?.message || 'WhatsApp template sending failed'
    )
  }

  return result
}
